### Case 2, Web analytics, PostHog style (`web-analytics`)

> **설명**: A date range drives every number. KPI cards with change badges, an area trend with a metric switch, source bars and a funnel from Box widths, a top-pages Table and a retention grid from @MixColor.

```js
// 최상위 루트 레이아웃: 전체 화면 구조와 주요 컴포넌트 배치
root = Col([
  head,
  kpis,
  trend,
  Grid([sources, pages], 2, 300, 4),
  Grid([funnel, retention], 2, 300, 4)
], 4)

// 상태 변수 ($range): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$range = "30d"

// 상태 변수 ($metric): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$metric = "Visitors"

// days 정의
days = $range == "7d" ? 7 : $range == "30d" ? 30 : 90

// traffic 정의
traffic = @Each(@Range(0, days), "i", {day: @FormatDate(@AddDays(@Today(), i - days + 1), "short"), Visitors: 1700 + 380 * ((i * 7) % 5) + i * 9 + (((i + 2) % 7) < 2 ? -520 : 0), Pageviews: 4200 + 820 * ((i * 3) % 4) + i * 21 + (((i + 2) % 7) < 2 ? -1300 : 0)})

// visitors 정의
visitors = @Sum(traffic.Visitors)

// views 정의
views = @Sum(traffic.Pageviews)

// 헤더 영역: 제목과 설명 텍스트, 컨트롤 요소
head = Row([
  Col([Title("Web analytics", "xl"), Caption("yoursite.com · sample data")], 1),

// 원본/파생 데이터 목록
shares = [
  {name: "Google", icon: "search", share: 41},
  {name: "Direct", icon: "mouse-pointer-click", share: 24},
  {name: "Hacker News", icon: "newspaper", share: 15},

sources = Card([
  Title("Top sources", "sm"),
  Col(@Each(shares, "s", Col([Row([Icon(s.icon, "sm", "muted"), Text(s.name, "sm"), Spacer(), Text(@Format(visitors * s.share / 100, "compact", 1), "sm", null, "medium", false, false, true)], 2), Box([Box([], null, null, null, "chart1", "full", null, (s.share / 41 * 100) + "%", 6)], null, null, null, "surface", "full", null, "100%", 6)], 1)), 3)
], null, 5, 3)

// pageRows 정의
pageRows = [
  {path: "/", share: 31, time: "0:48"},
  {path: "/pricing", share: 18, time: "1:52"},
  {path: "/blog/launch-week", share: 14, time: "4:05"},
```

### Case 6, Split the bill (`bill`)

> **설명**: Add and remove people and items (@Append, @Remove); who shared what with @Update and @Toggle; tax and tip; @Allocate keeps the cents adding up; copy the split.
> **사용자 프롬프트**: "Make a bill splitter for dinner with friends"

```js
// 최상위 루트 레이아웃: 전체 화면 구조와 주요 컴포넌트 배치
root = Card([
  head,
  people,
  addPerson,
  Divider(),
  itemsHead,
  items,
  addItem,
  Divider(),
  rates,
  summary
], null, 5, 4)

// 상태 변수 ($mode): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$mode = "item"

// 상태 변수 ($people): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$people = ["You", "Alex", "Sam", "Jordan"]

// 상태 변수 ($items): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$items = [
  {name: "Burrata & sourdough", amount: 18, people: ["You", "Alex", "Sam", "Jordan"]},
  {name: "Mains", amount: 96, people: ["You", "Alex", "Sam", "Jordan"]},
  {name: "A bottle of red", amount: 42, people: ["You", "Alex", "Jordan"]}
]

// 상태 변수 ($name): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$name = ""

// 상태 변수 ($itemName): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$itemName = ""

// 상태 변수 ($itemAmount): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$itemAmount = null

// 상태 변수 ($tax): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$tax = 8.5

// 상태 변수 ($tip): 사용자 인터랙션에 따라 값이 변경되는 반응형 상태
$tip = 20

// 헤더 영역: 제목과 설명 텍스트, 컨트롤 요소
head = Row([
  Col([
    Title("The check, please", "lg"),
    Text($mode == "item" ? "Tap names on an item to say who shared it." : "Everyone pays the same share.", "sm", "muted")
  ], 1),
  Spacer(),
  SegmentedControl($mode, [{value: "item", label: "By item"}, {value: "equal", label: "Equally"}])
], 3, null, null, true)

// people 정의
people = Row(@Each($people, "p", Button(p, Action([@Set($people, @Remove($people, "", p)), @Set($items, @Each($items, "it", {name: it.name, amount: it.amount, people: @Remove(it.people, "", p)}))]), "soft", null, "sm", "x")), 2, null, null, true)

// addPerson 정의
addPerson = Row([
  Input($name, null, "text", "Add a person"),
  Button("Add", Action([@Set($people, @Append($people, $name)), @Set($items, @Each($items, "it", {name: it.name, amount: it.amount, people: @Append(it.people, $name)})), @Reset($name)]), "outline", null, null, "user-plus", $name == "" || @Includes($people, $name))
], 2)

// itemsHead 정의
itemsHead = Row([
  Title("Items", "md"),
  Spacer(),
  Text(@Format(subtotal, "currency"), "sm", "muted", "medium", false, false, true)
])

// 원본/파생 데이터 목록
items = Col(@Each($items, "it", Card([Row([Text(it.name, "sm", null, "semibold"), Spacer(), Text(@Format(it.amount, "currency"), "sm", null, "medium", false, false, true), IconButton("trash-2", @Set($items, @Remove($items, "name", it.name)), "Remove " + it.name, "ghost")], 2), $mode == "item" ? Row(@Each($people, "p", Button(p, @Set($items, @Update($items, "name", it.name, "people", @Toggle(it.people, p))), @Includes(it.people, p) ? "solid" : "outline", null, "sm", @Includes(it.people, p) ? "check" : null)), 1, null, null, true) : null], null, 3, 2)), 2)

// addItem 정의
addItem = Row([
  Input($itemName, null, "text", "Item"),
  Input($itemAmount, null, "number", "Price"),
  Button("Add item", Action([@Set($items, @Append($items, {name: $itemName, amount: $itemAmount, people: $people})), @Reset($itemName, $itemAmount)]), "outline", null, null, "plus", $itemName == "" || $itemAmount <= 0)
], 2)

// rates 정의
rates = Grid([
  Slider($tax, 0, 15, 0.5, "Tax", "%"),
  Slider($tip, 0, 30, 1, "Tip", "%")
], 2, 220, 6)

// subtotal 정의
subtotal = @Sum($items.amount)

// factor 정의
factor = 1 + ($tax + $tip) / 100

// unassigned 정의
unassigned = $mode == "equal" ? 0 : @Sum(@Each($items, "it", @Count(it.people) == 0 ? it.amount : 0))

// raw 정의
raw = @Each($people, "p", factor * ($mode == "equal" ? subtotal / @Max([@Count($people), 1]) : @Sum(@Each($items, "it", @Includes(it.people, p) ? it.amount / @Count(it.people) : 0))))

// cents 정의
cents = @Sum(raw) > 0 ? @Allocate(@Round(@Sum(raw), 2), raw) : raw

// 원본/파생 데이터 목록
shares = @Each(@Range(0, @Count($people)), "i", {name: $people[i], amount: cents[i]})

// 요약 및 결과 표시 카드
summary = Card([
  Col(@Each(shares, "s", Row([Text(s.name), Spacer(), Text(@Format(s.amount, "currency"), null, null, "semibold", false, false, true)])), 2),
  Divider(),
  Row([
    Text("Total with tax and tip", "sm", "muted"),
    Spacer(),
    Title(@Format(subtotal * factor, "currency"), "lg")
  ]),
  unassigned > 0 ? Text(@Format(unassigned, "currency") + " is not assigned to anyone yet.", "xs", "warning") : null,
  Button("Copy the split", @CopyToClipboard(@Join(@Each(shares, "s", s.name + ": " + @Format(s.amount, "currency")), "\n")), "solid", null, "sm", "copy")
], "soft", 4, 3)
```
